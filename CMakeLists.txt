cmake_minimum_required(VERSION 3.24)
project(xfrog VERSION 2.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

if(WIN32)
    include(FetchContent)
    set(SDL_SHARED OFF CACHE BOOL "" FORCE)
    set(SDL_STATIC ON CACHE BOOL "" FORCE)
    set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
    set(SDL_TESTS OFF CACHE BOOL "" FORCE)
    set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(SDL3
        URL https://github.com/libsdl-org/SDL/releases/download/release-3.4.18/SDL3-3.4.18.tar.gz
        URL_HASH SHA256=9c75cf16330322c217dedd2e0609f1124f1b54b8633e763467b4684d0f4334a3
        DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(SDL3)
else()
    find_package(SDL3 3.2 REQUIRED CONFIG)
endif()

add_library(xfrog-model STATIC src/v2/game.cpp)
target_compile_features(xfrog-model PUBLIC cxx_std_17)
target_include_directories(xfrog-model PUBLIC src/v2)
add_library(xfrog-presentation STATIC src/v2/render.cpp src/v2/audio.cpp)
target_compile_features(xfrog-presentation PUBLIC cxx_std_17)
target_include_directories(xfrog-presentation SYSTEM PRIVATE vendor)
target_link_libraries(xfrog-presentation PUBLIC xfrog-model SDL3::SDL3)
add_executable(xfrog WIN32 src/v2/main.cpp)
target_link_libraries(xfrog PRIVATE xfrog-presentation)
set_target_properties(xfrog PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin"
    VS_DEBUGGER_WORKING_DIRECTORY "$<TARGET_FILE_DIR:xfrog>")
add_custom_target(xfrog-assets ALL
    COMMAND ${CMAKE_COMMAND} "-DSOURCE_DIR=${CMAKE_CURRENT_SOURCE_DIR}"
        "-DOUTPUT_DIR=$<TARGET_FILE_DIR:xfrog>" -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/StageV2.cmake"
    VERBATIM)
add_dependencies(xfrog xfrog-assets)
if(MSVC)
    target_compile_options(xfrog-model PRIVATE /W4)
    target_compile_options(xfrog-presentation PRIVATE /W4)
else()
    target_compile_options(xfrog-model PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(xfrog-presentation PRIVATE -Wall -Wextra -Wpedantic)
endif()
option(XFROG_BUILD_CLASSIC "Also build the preserved Xfrog 1.03 implementation" OFF)
if(XFROG_BUILD_CLASSIC)
    include(cmake/Classic.cmake)
endif()
include(CTest)
if(BUILD_TESTING)
    add_subdirectory(tests/v2)
    if(XFROG_BUILD_CLASSIC AND NOT WIN32)
        add_subdirectory(tests/classic)
    endif()
endif()

option(XFROG_BUILD_ART_TOOLS "Build offline courier sprite authoring tool (SDL >= 3.4)" OFF)
if(XFROG_BUILD_ART_TOOLS)
    if(NOT WIN32)
        find_package(SDL3 3.4 REQUIRED CONFIG)
    endif()
    add_executable(xfrog-bake-courier EXCLUDE_FROM_ALL tools/bake_courier.cpp)
    target_link_libraries(xfrog-bake-courier PRIVATE xfrog-presentation)
endif()
